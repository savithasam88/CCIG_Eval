% C6 Existential-Universal: At least one candidate object (P1=V1) must relate by D1 to every target object (P2=V2).
%:- #count{X : object(X), hasProperty(X,P1,V1), #count{Y : object(Y), hasProperty(Y,P2,V2), not hasRelationship(X,Y,D1)} = 0} = 0.
c6_1_prop_counterexample(X) :- hasProperty(X, P1, V1), hasProperty(Y, P2, V2, not hasRelationship(X, Y, D1).
c6_1_prop_satisfied :- hasProperty(X, P1, V1), not c6_1_prop_counterexample(X).
:- not c6_1_prop_satisfied.
