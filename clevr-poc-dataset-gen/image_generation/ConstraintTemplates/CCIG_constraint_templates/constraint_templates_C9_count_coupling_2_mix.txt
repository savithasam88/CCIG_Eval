% C9 Aggregate Comparison: Requires equal cardinality between two independently filtered subsets.
ccig_coupling_left(N) :- N = #count { X : object(X), not hasProperty(X, P1', V1'), hasProperty(X, P2', V2'), P1'!=P2'}.
ccig_coupling_right(N) :- N = #count { Y : object(Y), not hasProperty(Y, P3', V3'), hasProperty(Y, P4', V4'), P4'!=P3'}.
:- ccig_coupling_left(N), ccig_coupling_right(M), N != M.
